// Creates a TCP socket
int create_socket() {
    int sock = socket(AF_INET, SOCK_STREAM, 0);
    if (sock == -1) {
        perror("Could not create socket");
        exit(EXIT_FAILURE);
    }
    // Allow reuse of address, so you can restart the server quickly
    int opt = 1;
    if (setsockopt(sock, SOL_SOCKET, SO_REUSEADDR, (char *)&opt, sizeof(opt)) < 0) {
        perror("setsockopt failed");
        exit(EXIT_FAILURE);
    }
    return sock;
}

// Binds the socket to a given port and sets it to listen for connections
void bind_socket(int server_fd, int port) {
    struct sockaddr_in server;
    server.sin_family = AF_INET;
    server.sin_addr.s_addr = INADDR_ANY; // Listen on all available network interfaces
    server.sin_port = htons(port);       // Convert port to network byte order

    if (bind(server_fd, (struct sockaddr *)&server, sizeof(server)) < 0) {
        perror("Bind failed");
        exit(EXIT_FAILURE);
    }

    listen(server_fd, 3); // Max 3 pending connections in queue
    printf("Server listening on port %d\n", port);
}

// Main server loop: accepts new client connections and creates a handler thread for each
void start_server(int server_fd) {
    int client_sock;
    struct sockaddr_in client_address;
    socklen_t client_len = sizeof(client_address);
    Client *new_client;

    while (1) {
        printf("Waiting for incoming connections...\n");
        client_sock = accept(server_fd, (struct sockaddr *)&client_address, &client_len);
        if (client_sock < 0) {
            perror("Accept failed");
            continue; // Continue waiting for other connections
        }
        printf("Connection accepted from %s:%d\n", inet_ntoa(client_address.sin_addr), ntohs(client_address.sin_port));

        new_client = create_client_struct(client_sock, &client_address);
        if (!new_client) {
            fprintf(stderr, "Failed to create client struct, closing socket.\n");
            close(client_sock);
            continue;
        }

        pthread_t client_thread;
        // Create a new thread for each client to handle their communication concurrently
        if (pthread_create(&client_thread, NULL, client_handler, (void *)new_client) != 0) {
            perror("Failed to create client handler thread");
            cleanup_client(new_client); // Free client resources if thread creation fails
            close(client_sock);
        } else {
            // Detach the thread so its resources are automatically reclaimed when it finishes
            // without requiring an explicit pthread_join in the main server loop.
            pthread_detach(client_thread);
        }
    }
}

// Initializes a Sale structure with given details
void initialize_sale(Sale *sale, int id, const char *name, int max_items_cap, int max_clients_cap, const char *multicast_ip_suffix) {
    sale->id = id;
    sale->name = strdup(name); // Duplicate name string
    if (!sale->name) {
        perror("Failed to allocate sale name");
        exit(EXIT_FAILURE);
    }

    sale->max_items_capacity = max_items_cap;
    sale->items = (Item **)calloc(max_items_cap, sizeof(Item *));
    if (!sale->items) {
        perror("Failed to allocate sale items array");
        exit(EXIT_FAILURE);
    }
    sale->num_items = 0;

    sale->max_clients_capacity = max_clients_cap;
    sale->clients = (Client **)calloc(max_clients_cap, sizeof(Client *));
    if (!sale->clients) {
        perror("Failed to allocate sale clients array");
        exit(EXIT_FAILURE);
    }
    sale->num_clients = 0;

    sprintf(sale->multicast_ip, "%s%s", MULTICAST_BASE, multicast_ip_suffix);
    pthread_mutex_init(&sale->lock, NULL);
    pthread_cond_init(&sale->cond, NULL);
    sale->sale_active = 0; // Not active by default, admin must start it
}

// Adds an item to a sale
void add_item_to_sale(Sale *sale, int item_id, const char *item_name, float start_price) {
    if (sale->num_items >= sale->max_items_capacity) {
        fprintf(stderr, "Sale %s (ID: %d) is at max item capacity.\n", sale->name, sale->id);
        return;
    }

    Item *item = (Item *)malloc(sizeof(Item));
    if (!item) {
        perror("Failed to allocate item");
        exit(EXIT_FAILURE);
    }
    item->id = item_id;
    item->name = strdup(item_name);
    if (!item->name) {
        perror("Failed to allocate item name");
        free(item);
        exit(EXIT_FAILURE);
    }
    item->start_price = start_price;
    item->current_highest_bid = start_price; // Initial bid is starting price
    item->highest_bidder = NULL;             // No bidder initially
    item->clients_responded_count = 0;       // Reset for each new item
    pthread_mutex_init(&item->item_lock, NULL);
    pthread_cond_init(&item->item_cond, NULL);

    pthread_mutex_lock(&sale->lock);
    sale->items[sale->num_items++] = item;
    pthread_mutex_unlock(&sale->lock);
    printf("Added item %s (ID: %d) to Sale %s (ID: %d)\n", item_name, item_id, sale->name, sale->id);
}

// Creates and initializes a new Client structure
Client *create_client_struct(int sock, struct sockaddr_in *address) {
    Client *client = (Client *)malloc(sizeof(Client));
    if (!client) {
        perror("Failed to allocate client struct");
        return NULL;
    }
    memset(client, 0, sizeof(Client)); // Initialize all members to 0/NULL

    client->sock = sock;
    client->address = *address;
    inet_ntop(AF_INET, &(address->sin_addr), client->ip_address, INET_ADDRSTRLEN);
    client->username = NULL; // Will be set during login
    client->password = NULL; // Will be set during login
    client->sale = NULL;     // Not in a sale initially
    client->connection_closed = 0;
    pthread_mutex_init(&client->client_lock, NULL); // Initialize client-specific mutex
    return client;
}

// Adds a client to a sale's client list
void add_client_to_sale(Sale *sale, Client *client) {
    pthread_mutex_lock(&sale->lock);
    if (sale->num_clients >= sale->max_clients_capacity) {
        fprintf(stderr, "Sale %s (ID: %d) is full, cannot add client %s.\n", sale->name, sale->id, client->username);
        send_general_message_stoc(client, "Failed to join sale: Sale is full.");
        pthread_mutex_unlock(&sale->lock);
        return;
    }
    // Check if client is already in this sale
    for (int i = 0; i < sale->num_clients; i++) {
        if (sale->clients[i] == client) {
            pthread_mutex_unlock(&sale->lock);
            return; // Client already present
        }
    }
    pthread_mutex_lock(&client->client_lock); // Lock client struct before modifying its sale pointer
    client->sale = sale; // Set the client's current sale
    pthread_mutex_unlock(&client->client_lock);

    sale->clients[sale->num_clients++] = client;
    printf("Client %s (IP: %s) joined Sale %s (ID: %d). Total clients in sale: %d\n",
           client->username, client->ip_address, sale->name, sale->id, sale->num_clients);
    pthread_cond_signal(&sale->cond); // Signal if any threads are waiting for clients to join
    pthread_mutex_unlock(&sale->lock);
}

// Removes a client from a sale's client list
void remove_client_from_sale(Sale *sale, Client *client) {
    pthread_mutex_lock(&sale->lock);
    for (int i = 0; i < sale->num_clients; i++) {
        if (sale->clients[i] == client) {
            sale->clients[i] = sale->clients[sale->num_clients - 1]; // Swap with last client
            sale->clients[sale->num_clients - 1] = NULL; // Clear last slot
            sale->num_clients--;
            pthread_mutex_lock(&client->client_lock); // Lock client's mutex
            client->sale = NULL; // Remove client's reference to sale
            pthread_mutex_unlock(&client->client_lock);
            printf("Client %s (IP: %s) left Sale %s (ID: %d). Remaining clients: %d\n",
                   client->username, client->ip_address, sale->name, sale->id, sale->num_clients);
            break;
        }
    }
    pthread_mutex_unlock(&sale->lock);
}

// Frees memory associated with a Client structure
void cleanup_client(Client *client) {
    if (!client) return;

    // Ensure client is removed from any sale it was part of before freeing
    // Lock client's mutex to safely read its sale pointer
    pthread_mutex_lock(&client->client_lock);
    if (client->sale) {
        remove_client_from_sale(client->sale, client); // This function will also lock/unlock sale->lock
    }
    pthread_mutex_unlock(&client->client_lock); // Unlock client's mutex after accessing sale

    if (client->username) free(client->username);
    if (client->password) free(client->password);
    pthread_mutex_destroy(&client->client_lock); // Destroy the mutex associated with the client
    printf("Cleaned up client struct for %s (IP: %s)\n", client->ip_address, client->username ? client->username : "N/A");
    free(client);
}

// Frees memory associated with a Sale structure and its items
void cleanup_sale(Sale *sale) {
    if (!sale) return;

    if (sale->name) free(sale->name);

    if (sale->items) {
        for (int i = 0; i < sale->num_items; i++) {
            if (sale->items[i]) {
                if (sale->items[i]->name) free(sale->items[i]->name);
                pthread_mutex_destroy(&sale->items[i]->item_lock);
                pthread_cond_destroy(&sale->items[i]->item_cond);
                free(sale->items[i]);
            }
        }
        free(sale->items);
    }
    if (sale->clients) {
        // Clients themselves are freed by their respective client_handler threads,
        // not by the sale cleanup. This just frees the array of pointers.
        free(sale->clients);
    }

    pthread_mutex_destroy(&sale->lock);
    pthread_cond_destroy(&sale->cond);
    printf("Cleaned up sale struct for Sale %s (ID: %d)\n", sale->name ? sale->name : "N/A", sale->id);
    // Do NOT free the Sale* itself here if it's part of a global array `sales`.
    // It will be part of the `sales` array, which is freed in main.
}

// CLIENT COMMUNICATION
// ========================

// Thread entry point for each connected client
void *client_handler(void *client_ptr) {
    Client *client = (Client *)client_ptr; // Cast the void pointer to Client pointer

    printf("Client handler started for %s:%d (sock %d)\n", client->ip_address, ntohs(client->address.sin_port), client->sock);

    // This loop handles all messages for this specific client until connection is closed
    handle_client_messages(client);

    // Cleanup resources for this client when the message loop ends
    printf("Client handler for %s:%d (sock %d) is exiting.\n", client->ip_address, ntohs(client->address.sin_port), client->sock);
    close_client_connection(client); // Ensure socket is closed and flags are set
    cleanup_client(client);          // Free dynamically allocated client struct and its members

    pthread_exit(NULL); // Terminate the client handler thread
}

// Handles message reception and processing for a client
void handle_client_messages(Client *client) {
    char buffer[BUFFER_SIZE];
    ssize_t bytes_received;

    while (!client->connection_closed) { // Keep receiving as long as connection is active
        bytes_received = recv(client->sock, buffer, BUFFER_SIZE - 1, 0);

        if (bytes_received > 0) {
            buffer[bytes_received] = '\0'; // Null-terminate the received data
            printf("Received from %s (sock %d): %s\n", client->ip_address, client->sock, buffer);
            process_message(client, buffer); // Process the received message
        } else if (bytes_received == 0) {
            printf("Client %s (sock %d) disconnected gracefully.\n", client->ip_address);
            client->connection_closed = 1; // Set flag to exit loop
        } else {
            // Check for non-blocking errors (EAGAIN/EWOULDBLOCK) or bad file descriptor (EBADF/ENOTSOCK)
            // if socket was closed by another thread (e.g., client exited with -1).
            if (errno == EAGAIN || errno == EWOULDBLOCK) {
                usleep(10000); // Small delay to prevent busy-waiting
                continue;
            } else if (errno == EBADF || errno == ENOTSOCK) {
                printf("Client %s (sock %d): Socket already closed by another action.\n", client->ip_address, client->sock);
                client->connection_closed = 1; // Exit loop if socket is bad/not socket
            }
            else {
                perror("recv failed");
                client->connection_closed = 1; // Set flag to exit loop on other errors
            }
        }
    }
}

// Closes socket and flags client as closed
void close_client_connection(Client *client) {
    pthread_mutex_lock(&client->client_lock);
    if (!client->connection_closed) { // Only close if not already closed
        printf("Closing socket %d for client %s.\n", client->sock, client->ip_address);
        close(client->sock);
        client->connection_closed = 1; // Mark as closed
    }
    pthread_mutex_unlock(&client->client_lock);
}

// Closes all client connections within a specific sale
void close_client_connections_in_sale(Sale *sale) {
    pthread_mutex_lock(&sale->lock);
    for (int i = 0; i < sale->num_clients; i++) {
        if (sale->clients[i] && !sale->clients[i]->connection_closed) {
            send_general_message_stoc(sale->clients[i], "Sale has ended. Disconnecting.");
            close_client_connection(sale->clients[i]); // Closes socket and marks as closed
        }
    }
    // Clients will be cleaned up by their individual client_handler threads
    pthread_mutex_unlock(&sale->lock);
}

// Dispatches incoming messages to the appropriate handler based on message type
void process_message(Client *client, const char *message) {
    if (!message || strlen(message) == 0) {
        printf("Empty or null message received from %s\n", client->ip_address);
        return;
    }

    int message_type = message[0] - '0'; // Convert first character (type) to integer
    printf("Processing message type %d from client %s (Message: %s)\n", message_type, client->username ? client->username : client->ip_address, message);  // Added full message for debugging

    switch (message_type) {
        case 1: process_welcome_message(client, message); break; // Client login
        case 4: process_sale_ctos_message(client, message); break; // Client join/exit sale
        case 6: process_item_ctos_message(client, message); break; // Client bid on item
        case 8: process_keep_alive_message(client, message); break; // Client keep-alive
        default:
            fprintf(stderr, "Unknown message type %d received from client %s.\n", message_type, client->username ? client->username : client->ip_address);
            send_general_message_stoc(client, "Unknown command.");
            break;
    }
}

// Handles client login (type 1) and sends list of sales (type 9)
void process_welcome_message(Client *client, const char *message) {
    int username_len = atoi(strndup(message + 1, 2));
    int password_len = atoi(strndup(message + 3, 2));

    if (username_len <= 0 || password_len <= 0 || (1 + 2 + 2 + username_len + password_len) > strlen(message)) {
        send_general_message_stoc(client, "Login failed: Invalid format.");
        return;
    }

    // Allocate memory for username and password
    char *username = strndup(message + 5, username_len);
    char *password = strndup(message + 5 + username_len, password_len);

    if (!username || !password) {
        perror("Memory allocation failed for username/password");
        if (username) free(username);
        send_general_message_stoc(client, "Login failed: Server error.");
        return;
    }

    pthread_mutex_lock(&client->client_lock);
    // Store username and password (for simplicity, no actual authentication DB)
    if (client->username) free(client->username);
    if (client->password) free(client->password);
    client->username = username; // Assign, not strdup again
    client->password = password; // Assign, not strdup again
    pthread_mutex_unlock(&client->client_lock);

    printf("Client %s (IP: %s) logged in with password %s.\n", client->username, client->ip_address, client->password);

    // Simple authentication logic
    if (strcmp(client->password, SECRET_PASSWORD) == 0) { // Using SECRET_PASSWORD for authentication
        send_general_message_stoc(client, "Login successful. Welcome!");
        sleep(1); // Give client time to receive welcome
        send_list_of_sales(client); // After successful login, send available sales
    } else {
        send_general_message_stoc(client, "Login failed: Incorrect password.");
        close_client_connection(client); // Close connection on failed login
        // pthread_exit(NULL); // Do not exit thread here, let client_handler cleanup
    }
}

// Handles client requests to join/exit a sale (type 4)
void process_sale_ctos_message(Client *client, const char *message) {
    int sale_id_len = atoi(strndup(message + 1, 2));
    if (sale_id_len <= 0 || (1 + 2 + sale_id_len) > strlen(message)) {
        send_general_message_stoc(client, "Invalid sale ID format.");
        fprintf(stderr, "Server: Malformed Type 4 message from %s: %s\n", client->ip_address, message);
        return;
    }

    char *sale_id_str = strndup(message + 3, sale_id_len);
    int sale_id = atoi(sale_id_str);
    free(sale_id_str);

    if (sale_id == -1) { // Client wants to exit current sale
        if (client->sale) {
            printf("Client %s (User: %s) chose to exit Sale %d.\n", client->ip_address, client->username, client->sale->id);
            send_general_message_stoc(client, "Leaving current sale.");
            remove_client_from_sale(client->sale, client);
            // client->sale is set to NULL within remove_client_from_sale
        } else {
            send_general_message_stoc(client, "You are not currently in a sale.");
        }
        send_list_of_sales(client); // After exiting (or if not in a sale), send list of sales again
        return;
    }

    // Client wants to join a sale
    if (sale_id >= 0 && sale_id < num_sales) { // Sales IDs are 0-indexed on server side
        Sale *target_sale = &sales[sale_id];
        pthread_mutex_lock(&target_sale->lock);
        // Check if sale is active before allowing join
        if (!target_sale->sale_active) {
            send_general_message_stoc(client, "Sale is not active yet. Please wait or choose another.");
            pthread_mutex_unlock(&target_sale->lock);
            send_list_of_sales(client); // Re-send list of sales
            return;
        }
        pthread_mutex_unlock(&target_sale->lock);

        // If already in this sale, do nothing
        if (client->sale == target_sale) {
            send_general_message_stoc(client, "You are already in this sale.");
            return;
        }

        // If in a different sale, leave it first
        if (client->sale) {
            send_general_message_stoc(client, "Leaving previous sale to join a new one.");
            remove_client_from_sale(client->sale, client);
        }

        add_client_to_sale(target_sale, client);
        printf("Client %s (User: %s) joined Sale ID %d\n", client->ip_address, target_sale->id);
        send_sale_stoc_message(client, target_sale); // Send client details of the joined sale
    } else {
        send_general_message_stoc(client, "Invalid sale ID. Please choose an existing sale.");
        send_list_of_sales(client); // Re-send list of sales
    }
}

// Handles client bid for item (type 6)
void process_item_ctos_message(Client *client, const char *message) {
    if (!client->sale || !client->sale->sale_active) {
        send_general_message_stoc(client, "You must join an active sale before placing bids.");
        return;
    }

    int item_id_len = atoi(strndup(message + 1, 2));
    int price_len = atoi(strndup(message + 3, 2)); // FIX: Client sends 2 digits for price_len

    if (item_id_len <= 0 || price_len <= 0 || (1 + 2 + 2 + item_id_len + price_len) > strlen(message)) {
        send_general_message_stoc(client, "Invalid bid format received by server.");
        fprintf(stderr, "Server: Malformed Type 6 message from %s: %s\n", client->ip_address, message);
        return;
    }

    char *item_id_str = strndup(message + 5, item_id_len);
    char *price_str = strndup(message + 5 + item_id_len, price_len);

    int item_id = atoi(item_id_str);
    float bid_price = atof(price_str);

    free(item_id_str);
    free(price_str);

    Item *target_item = NULL;
    pthread_mutex_lock(&client->sale->lock); // Lock sale to safely access items list
    for (int i = 0; i < client->sale->num_items; i++) {
        if (client->sale->items[i] != NULL && client->sale->items[i]->id == item_id) {
            target_item = client->sale->items[i];
            break;
        }
    }
    pthread_mutex_unlock(&client->sale->lock);

    if (!target_item) {
        send_general_message_stoc(client, "Item not found in your current sale or bidding has closed for it.");
        return;
    }

    // Process bid
    pthread_mutex_lock(&target_item->item_lock); // Lock item to safely update bid
    if (bid_price > target_item->current_highest_bid) {
        target_item->current_highest_bid = bid_price;
        target_item->highest_bidder = client;
        printf("Client %s (User: %s) bid %.2f on Item %d (%s). New highest bid.\n",
               client->ip_address, client->username, bid_price, target_item->id, target_item->name);
        send_general_message_stoc(client, "Your bid is now the highest!");
        // Optionally, send a general message to all other clients in the sale about the new highest bid
        char update_msg[BUFFER_SIZE];
        sprintf(update_msg, "Item %d: New highest bid is %.2f by %s!", target_item->id, bid_price, client->username);
        send_general_message_to_all(client->sale, update_msg);
    } else {
        printf("Client %s (User: %s) bid %.2f on Item %d (%s). Bid too low.\n",
               client->ip_address, client->username, bid_price, target_item->id, target_item->name);
        send_general_message_stoc(client, "Your bid was too low. Please bid higher.");
    }
    target_item->clients_responded_count++; // Increment responded count for this item
    pthread_cond_signal(&target_item->item_cond); // Signal any threads waiting on item responses
    pthread_mutex_unlock(&target_item->item_lock);
}

// Sends item details (type 5) to a single client
void send_item_to_client(Client *client, Item *item) {
    if (!client || client->sock <= 0 || client->connection_closed || !item || !item->name) {
        fprintf(stderr, "Attempted to send item details to invalid client or for invalid item.\n");
        return;
    }
    char message[BUFFER_SIZE];
    // Format: "5" + 2-digit item_id_len + 2-digit item_name_len + 2-digit price_len + item_id + item_name + start_price
    char item_id_str[10]; sprintf(item_id_str, "%d", item->id);
    // Use current_highest_bid for display to client
    char item_price_str[20]; sprintf(item_price_str, "%.2f", item->current_highest_bid);

    sprintf(message, "5%02zu%02zu%02zu%s%s%s",
            strlen(item_id_str), strlen(item->name), strlen(item_price_str),
            item_id_str, item->name, item_price_str);
    send_message(client->sock, message);
}

// Sends winner packet (type 7) to a single client
void send_winner_packet(Client *client, int item_id, const char *winner_name, float winning_price) {
    if (!client || client->sock <= 0 || client->connection_closed || !winner_name) {
        fprintf(stderr, "Attempted to send winner packet to invalid client.\n");
        return;
    }
    char message[BUFFER_SIZE];
    char item_id_str[10]; sprintf(item_id_str, "%d", item_id);
    char price_str[20]; sprintf(price_str, "%.2f", winning_price);

    // Format: "7" + 2-digit item_id_len + 2-digit winner_name_len + 2-digit price_len + item_id + winner_name + winning_price
    // FIX: price_len is now 2 digits (%02zu) to match client parsing
    sprintf(message, "7%02zu%02zu%02zu%s%s%s",
            strlen(item_id_str), strlen(winner_name), strlen(price_str),
            item_id_str, winner_name, price_str);
    send_message(client->sock, message);
}

// Sends winner packet to all clients in a sale
void send_winner_packet_to_all_in_sale(Sale *sale, int item_id, const char *winner_name, float winning_price) {
    if (!sale) return;
    pthread_mutex_lock(&sale->lock);
    for (int i = 0; i < sale->num_clients; i++) {
        if (sale->clients[i] && !sale->clients[i]->connection_closed) {
            send_winner_packet(sale->clients[i], item_id, winner_name, winning_price);
        }
    }
    pthread_mutex_unlock(&sale->lock);
}

    offset += sprintf(sales_list + offset, "Available Sales:\n");
    pthread_mutex_lock(&sales_list_lock); // Lock global sales array
    for (int i = 0; i < num_sales; i++) {
        // Lock individual sale before accessing its members for printing
        pthread_mutex_lock(&sales[i].lock);
        if (sales[i].name) { // Check if sale name is valid (i.e., sale struct is initialized)
            offset += sprintf(sales_list + offset, "  ID: %d, Name: %s (Multicast: %s) - %s. Clients: %d/%d. Items: %d/%d\n",
                              sales[i].id, sales[i].name, sales[i].multicast_ip,
                              sales[i].sale_active ? "Active" : "Inactive",
                              sales[i].num_clients, sales[i].max_clients_capacity,
                              sales[i].num_items, sales[i].max_items_capacity);
        }
        pthread_mutex_unlock(&sales[i].lock); // Unlock individual sale
    }
    pthread_mutex_unlock(&sales_list_lock); // Unlock global sales array

}

// Sends a UDP multicast message to a specific multicast IP
void send_multicast_message(const char *multicast_ip, const char *message) {
    int sock;
    struct sockaddr_in multicast_addr;

    if ((sock = socket(AF_INET, SOCK_DGRAM, 0)) < 0) {
        perror("Multicast socket creation failed");
        return;
    }

    memset(&multicast_addr, 0, sizeof(multicast_addr));
    multicast_addr.sin_family = AF_INET;
    multicast_addr.sin_addr.s_addr = inet_addr(multicast_ip);
    multicast_addr.sin_port = htons(SERVER_PORT); // Use same port as TCP for simplicity (or MULTICAST_PORT if server also listens)

    // Set Time-To-Live (TTL) for multicast packets (important for routing)
    int ttl = 1; // 1 means it stays on the local network segment
    if (setsockopt(sock, IPPROTO_IP, IP_MULTICAST_TTL, &ttl, sizeof(ttl)) < 0) {
        perror("setsockopt IP_MULTICAST_TTL failed");
        close(sock);
        return;
    }

    if (sendto(sock, message, strlen(message), 0, (struct sockaddr *)&multicast_addr, sizeof(multicast_addr)) < 0) {
        perror("Multicast sendto failed");
    } else {
        printf("Sent multicast to %s: %s\n", multicast_ip, message);
    }

    close(sock);
}

// Starts a sale in a new thread
void start_sale_in_thread(int sale_index) {
    if (sale_index < 0 || sale_index >= num_sales) {
        printf("Invalid sale index for starting.\n");
        return;
    }

    Sale *sale = &sales[sale_index];
    pthread_mutex_lock(&sale->lock);
    if (sale->sale_active) {
        printf("Sale %s (ID: %d) is already active.\n", sale->name, sale->id);
        pthread_mutex_unlock(&sale->lock);
        return;
    }
    sale->sale_active = 1; // Mark sale as active
    pthread_mutex_unlock(&sale->lock);

    start_sale_args *args = (start_sale_args *)malloc(sizeof(start_sale_args));
    if (!args) {
        perror("Failed to allocate start_sale_args");
        return;
    }
    args->sale_index = sale_index;

    pthread_t sale_thread;
    if (pthread_create(&sale_thread, NULL, start_sale_thread, (void *)args) != 0) {
        perror("Failed to create sale thread");
        free(args); // Free args if thread creation fails
    } else {
        pthread_detach(sale_thread); // Detach sale thread
        printf("Sale %s (ID: %d) started in a new thread.\n", sale->name, sale->id);
    }
}

// Thread entry point for managing a sale's items and bidding
void *start_sale_thread(void *arg) {
    start_sale_args *args = (start_sale_args *)arg;
    Sale *current_sale = &sales[args->sale_index];
    free(args); // Free the arguments structure

    printf("Sale thread for Sale %s (ID: %d) started.\n", current_sale->name, current_sale->id);
    send_general_message_to_all(current_sale, "The auction sale is starting now!");

    // Iterate through each item in the sale
    for (int i = 0; i < current_sale->num_items; i++) {
        Item *item = current_sale->items[i];
        if (!item) continue;

        printf("Auctioning Item %d: %s (Starting Price: %.2f) in Sale %s.\n",
               item->id, item->name, item->start_price, current_sale->name);

        pthread_mutex_lock(&item->item_lock);
        item->current_highest_bid = item->start_price; // Reset bid for new item
        item->highest_bidder = NULL;
        item->clients_responded_count = 0; // Reset response count
        pthread_mutex_unlock(&item->item_lock);

        // Announce the item via multicast to all clients in this sale
        char multicast_msg[BUFFER_SIZE];
        sprintf(multicast_msg, "NEW_ITEM: ID=%d, Name=%s, CurrentBid=%.2f",
                item->id, item->name, item->current_highest_bid);
        send_multicast_message(current_sale->multicast_ip, multicast_msg);

        // Instruct clients to send bids (TCP message)
        pthread_mutex_lock(&current_sale->lock);
        int num_clients_in_sale = current_sale->num_clients; // Get current client count under lock
        for (int k = 0; k < num_clients_in_sale; k++) {
            if (current_sale->clients[k] && !current_sale->clients[k]->connection_closed) {
                send_item_to_client(current_sale->clients[k], item); // Send type 5 message
            }
        }
        pthread_mutex_unlock(&current_sale->lock);

        printf("Waiting for %d seconds for bids on Item %d (%s)...\n", ITEM_BID_TIMEOUT_SECONDS, item->id, item->name);
        // This is a simple blocking sleep. A more advanced system might use
        // pthread_cond_timedwait on item->item_cond to wait for all clients
        // to respond or timeout, then collect final bids.
        sleep(ITEM_BID_TIMEOUT_SECONDS);

        // Determine winner after timeout
        pthread_mutex_lock(&item->item_lock);
        if (item->highest_bidder) {
            printf("Item %d (%s) winner: %s with a bid of %.2f\n",
                   item->id, item->name, item->highest_bidder->username, item->current_highest_bid);
            send_winner_packet_to_all_in_sale(current_sale, item->id, item->highest_bidder->username, item->current_highest_bid);
        } else {
            printf("No bids received for Item %d (%s). Item unsold.\n", item->id, item->name);
            send_general_message_to_all(current_sale, "No bids received for the last item. Item unsold.");
        }
        pthread_mutex_unlock(&item->item_lock);

        // Short pause between items
        sleep(2);
    }

    printf("All items in Sale %s (ID: %d) have been auctioned.\n", current_sale->name, current_sale->id);
    send_general_message_to_all(current_sale, "The auction sale has finished!");
    end_sale(current_sale); // End and cleanup the sale
    pthread_exit(NULL); // Terminate the sale thread
}

// Ends a sale: notifies clients, cleans up sale resources, and removes it from global list
void end_sale(Sale *sale) {
    if (!sale) return;

    pthread_mutex_lock(&sale->lock);
    sale->sale_active = 0; // Mark sale as inactive
    pthread_mutex_unlock(&sale->lock);

    // Close client connections that were part of this sale (they will also self-cleanup)
    close_client_connections_in_sale(sale);

    // Remove the sale from the global sales list (this will call cleanup_sale)
    remove_sale_from_list(sale->id);
}

// Removes a sale from the global sales list and cleans up its memory
void remove_sale_from_list(int sale_id_to_remove) {
    pthread_mutex_lock(&sales_list_lock);
    for (int i = 0; i < num_sales; i++) {
        if (sales[i].id == sale_id_to_remove) {
            printf("Removing Sale %d (%s) from global list.\n", sales[i].id, sales[i].name);
            cleanup_sale(&sales[i]); // Clean up its internal resources (items, arrays, mutexes)

            // Shift remaining sales to fill the gap
            for (int j = i; j < num_sales - 1; j++) {
                sales[j] = sales[j + 1];
            }
            num_sales--; // Decrement the total count of sales
            break;
        }
    }
    pthread_mutex_unlock(&sales_list_lock);
    // Note: The global `sales` array is allocated in main and freed there.
    // If you need to dynamically resize `sales` itself, you'd use realloc.
}

    while (1) {                // Enters an infinite loop to continuously accept commands
        printf("\nEnter command (help, list_sales, start_sale <ID>, check_ip <IP>, Kofiko): "); // Prompts the admin to enter a command

            break; // Exit loop on error
        }

        if (strcmp(command, "help") == 0) { // If the command is "help"

        } else if (strcmp(command, "list_sales") == 0) { // If the command is "list_sales"
            print_all_sales();              // Calls a function to display details of all active sales
        } else if (strncmp(command, "start_sale ", 11) == 0) { // If the command starts with "start_sale "
            int sale_id = atoi(command + 11); // Extracts the sale ID from the command string
            pthread_mutex_lock(&sales_list_lock); // Lock before accessing global sales array
            if (sale_id >= 0 && sale_id < num_sales) { // Check if sale_id is valid based on `num_sales`
                start_sale_in_thread(sale_id); // Starts the specified sale in a new thread
            } else {
                printf("Invalid sale ID. Must be between 0 and %d.\n", num_sales - 1); // Prints an error if the sale ID is invalid
            }
            pthread_mutex_unlock(&sales_list_lock);
        } else if (strncmp(command, "check_ip ", 9) == 0) { // If the command starts with "check_ip "
            char ip_to_check[INET_ADDRSTRLEN];
            // Copies the IP address part from the command
            strncpy(ip_to_check, command + 9, sizeof(ip_to_check) - 1);

        } else {
            printf("Unknown command. Type 'help' for available commands.\n"); // Informs if the command is not recognized
        }
    }
    return NULL; // Returns NULL (thread function signature requires a void* return)
}

    generate_multicast_ip(2, multicast_ip_buffer);
    initialize_sale(&sales[2], 2, "Jewelry Auction", max_items, max_clients, multicast_ip_buffer + strlen(MULTICAST_BASE));
    add_item_to_sale(&sales[2], 301, "Diamond Ring", 5000.00);
    add_item_to_sale(&sales[2], 302, "Gold Necklace", 1000.00);

    printf("Initialized %d sales.\n", num_sales);
}

//========================
// MAIN FUNCTION
//========================
int main() {
    // Set global capacities for sales
    max_items = 10; // Each sale can hold up to 10 items
    max_clients = 10; // Each sale can hold up to 10 clients

    int server_fd = create_socket();    // Create the main server listening socket
    bind_socket(server_fd, SERVER_PORT); // Bind the socket to the defined port

    // This creates a dedicated thread for the server administrator to type commands.
    // It runs concurrently with the main server logic (accepting client connections, handling sales).
    pthread_t cmd_thread;
    if (pthread_create(&cmd_thread, NULL, command_handler, NULL) != 0) {
        perror("Failed to create command handler thread");
        exit(EXIT_FAILURE); // Server cannot function without command handler
    }
    pthread_detach(cmd_thread); // Detach admin command thread

    make_real_sales(); // Initializes some predefined auction sales on the server

    // Start the main server loop which accepts incoming client connections.
    // This function will run indefinitely, creating new threads for each client.
    start_server(server_fd);

    // This part of the code is effectively unreachable because `start_server` is an infinite loop.

    // Cleanup - Frees memory allocated for global sales array and its contents when the server exits.
    // This typically happens only when the process is terminated, as start_server is an infinite loop.
    printf("Server shutting down. Cleaning up resources...\n");
    pthread_mutex_lock(&sales_list_lock); // Lock global sales list before cleanup
    for (int i = 0; i < num_sales; i++) {
        cleanup_sale(&sales[i]); // Clean up individual sale resources
    }
    if (sales) {
        free(sales); // Free the main array of Sale structures
        sales = NULL;
    }
    pthread_mutex_unlock(&sales_list_lock);
    pthread_mutex_destroy(&sales_list_lock);

    close(server_fd); // Close the main server listening socket
    return 0;         // Exits the program successfully
}
